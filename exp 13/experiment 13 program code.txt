
#include <Arduino.h>

// FreeRTOS communication handles
QueueHandle_t messageQueue;
QueueHandle_t mailbox;
StreamBufferHandle_t streamBuffer;

// Producer Task
void ProducerTask(void *pvParameters)
{
  int data = 0;

  while (1)
  {
    data++;

    // Send data using Message Queue
    xQueueSend(messageQueue, &data, portMAX_DELAY);

    Serial.print("Producer: Queue Data = ");
    Serial.println(data);

    // Send latest data using Mailbox
    xQueueOverwrite(mailbox, &data);

    // Send data using Stream Buffer
    xStreamBufferSend(streamBuffer, &data,
                      sizeof(data), portMAX_DELAY);

    Serial.print("Producer: Produced Data = ");
    Serial.println(data);

    vTaskDelay(pdMS_TO_TICKS(1000));
  }
}

// Consumer Task
void ConsumerTask(void *pvParameters)
{
  int receivedData;

  while (1)
  {
    // Receive data from Message Queue
    if (xQueueReceive(messageQueue, &receivedData,
                      portMAX_DELAY) == pdPASS)
    {
      Serial.print("Consumer: Queue Data = ");
      Serial.println(receivedData);
    }

    vTaskDelay(pdMS_TO_TICKS(500));
  }
}

// Mailbox Consumer Task
void MailboxTask(void *pvParameters)
{
  int receivedData;

  while (1)
  {
    if (xQueueReceive(mailbox, &receivedData,
                      portMAX_DELAY) == pdPASS)
    {
      Serial.print("Mailbox Data = ");
      Serial.println(receivedData);
    }

    vTaskDelay(pdMS_TO_TICKS(500));
  }
}

// Stream Buffer Consumer Task
void StreamTask(void *pvParameters)
{
  int receivedData;

  while (1)
  {
    size_t bytesReceived =
      xStreamBufferReceive(streamBuffer, &receivedData,
                           sizeof(receivedData),
                           portMAX_DELAY);

    if (bytesReceived == sizeof(receivedData))
    {
      Serial.print("Stream Buffer Data = ");
      Serial.println(receivedData);
    }
  }
}

void setup()
{
  Serial.begin(115200);

  // Create Message Queue
  messageQueue = xQueueCreate(5, sizeof(int));

  // Create Mailbox (Queue of length 1)
  mailbox = xQueueCreate(1, sizeof(int));

  // Create Stream Buffer
  streamBuffer = xStreamBufferCreate(100, sizeof(int));

  if (messageQueue == NULL ||
      mailbox == NULL ||
      streamBuffer == NULL)
  {
    Serial.println("Communication Object Creation Failed!");
    while (1);
  }

  // Create FreeRTOS Tasks
  xTaskCreate(ProducerTask, "Producer", 2048, NULL, 2, NULL);
  xTaskCreate(ConsumerTask, "Consumer", 2048, NULL, 1, NULL);
  xTaskCreate(MailboxTask, "Mailbox", 2048, NULL, 1, NULL);
  xTaskCreate(StreamTask, "Stream", 2048, NULL, 1, NULL);

  Serial.println("Producer-Consumer Communication Started");
}

void loop()
{
  vTaskDelay(pdMS_TO_TICKS(1000));
}
